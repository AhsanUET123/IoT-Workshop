const int soilPin = A0;
const int greenLED = 8;
const int redLED = 9;

int moistureValue = 0;
bool isWet = false;

void setup() {

  Serial.begin(9600);

  pinMode(greenLED, OUTPUT);
  pinMode(redLED, OUTPUT);

  // Initially DRY
  digitalWrite(greenLED, LOW);
  digitalWrite(redLED, HIGH);

  Serial.println("Plant Monitoring System Started");
}

void loop() {

  long total = 0;

  // Average 10 readings
  for (int i = 0; i < 10; i++) {
    total += analogRead(soilPin);
    delay(20);
  }

  moistureValue = total / 10;

  Serial.print("Moisture Value: ");
  Serial.println(moistureValue);

  if (moistureValue < 400) {
    isWet = true;
  }
  else if (moistureValue > 500) {
    isWet = false;
  }

  if (isWet) {

    digitalWrite(greenLED, HIGH);
    digitalWrite(redLED, LOW);

    Serial.println("Soil Condition: WET");
  }
  else {

    digitalWrite(greenLED, LOW);
    digitalWrite(redLED, HIGH);

    Serial.println("Soil Condition: DRY");
  }

  Serial.println("--------------------------");

  delay(500);
}
